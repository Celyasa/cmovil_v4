import 'package:cmovil_v4/app/core/error/api_failure.dart';
import 'package:dio/dio.dart';

extension DioExceptionMapper on DioException {
  /// Traduce un error de Dio a un [ApiFailure].
  /// Uso en los data sources:
  ///   try { ... } on DioException catch (e) { throw e.toFailure(); }
  ApiFailure toFailure() {
    final status = response?.statusCode;

    switch (type) {
      case DioExceptionType.connectionTimeout:
      case DioExceptionType.sendTimeout:
      case DioExceptionType.receiveTimeout:
        return const ApiFailure(
          type: ApiFailureType.timeout,
          message: 'El servidor tardó demasiado en responder.',
        );
      case DioExceptionType.connectionError:
        return const ApiFailure(
          type: ApiFailureType.network,
          message: 'Sin conexión. Verifica tu red e inténtalo de nuevo.',
        );
      case DioExceptionType.cancel:
        return const ApiFailure(
          type: ApiFailureType.cancelled,
          message: 'La solicitud fue cancelada.',
        );
      case DioExceptionType.badResponse:
        return _fromStatus(status);
      case DioExceptionType.badCertificate:
      case DioExceptionType.unknown:
        return const ApiFailure(
          type: ApiFailureType.unknown,
          message: 'Ocurrió un error inesperado.',
        );
    }
  }

  ApiFailure _fromStatus(int? status) {
    final serverMessage = _serverMessage();

    return switch (status) {
      401 => ApiFailure(
          type: ApiFailureType.unauthorized,
          message: serverMessage ?? 'Credenciales o sesión inválidas.',
          statusCode: status,
        ),
      403 => ApiFailure(
          type: ApiFailureType.forbidden,
          message: serverMessage ?? 'No tienes permisos para esta acción.',
          statusCode: status,
        ),
      404 => ApiFailure(
          type: ApiFailureType.notFound,
          message: serverMessage ?? 'Recurso no encontrado.',
          statusCode: status,
        ),
      final int code when code >= 500 => ApiFailure(
          type: ApiFailureType.server,
          message: 'El servidor presentó un problema. Inténtalo más tarde.',
          statusCode: code,
        ),
      _ => ApiFailure(
          type: ApiFailureType.unknown,
          message: serverMessage ?? 'No se pudo completar la solicitud.',
          statusCode: status,
        ),
    };
  }

  /// Extrae `message` del cuerpo si la API lo envía como JSON.
  String? _serverMessage() {
    final data = response?.data;
    if (data is Map<String, dynamic>) {
      final message = data['message'];
      if (message is String && message.trim().isNotEmpty) return message;
    }
    return null;
  }
}
